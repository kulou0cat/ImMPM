cmake_minimum_required(VERSION 3.20)
project(ImplicitMPM2D LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(EIGEN_INCLUDE_DIR "D:/games101/eigen-3.4.0")
find_package(Eigen3 3.3 QUIET NO_MODULE)
if(NOT TARGET Eigen3::Eigen)
  find_path(EIGEN_HEADERS Eigen/Core HINTS "${EIGEN_INCLUDE_DIR}" PATH_SUFFIXES eigen3)
  if(NOT EIGEN_HEADERS)
    message(FATAL_ERROR "Set EIGEN_INCLUDE_DIR to the folder containing Eigen/Core")
  endif()
  add_library(Eigen3::Eigen INTERFACE IMPORTED)
  set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${EIGEN_HEADERS}")
endif()
set(OpenCV_DIR "D:/games101/opencv/build")
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui imgcodecs)
add_library(mpm_core src/MPM.cpp)
target_include_directories(mpm_core PUBLIC src)
target_link_libraries(mpm_core PUBLIC Eigen3::Eigen)
add_executable(ImplicitMPM2D src/main.cpp)
target_include_directories(ImplicitMPM2D PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ImplicitMPM2D PRIVATE mpm_core ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(mpm_core PRIVATE /utf-8 /W4)
  target_compile_options(ImplicitMPM2D PRIVATE /utf-8 /W4)
endif()
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT ImplicitMPM2D)
# Official Windows OpenCV packages normally expose imported library locations.
foreach(lib IN LISTS OpenCV_LIBS)
  if(TARGET ${lib})
    get_target_property(libtype ${lib} TYPE)
    if(libtype STREQUAL "SHARED_LIBRARY")
      add_custom_command(TARGET ImplicitMPM2D POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:${lib}>" "$<TARGET_FILE_DIR:ImplicitMPM2D>" VERBATIM)
    endif()
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(mpm_tests tests/test_core.cpp)
  target_link_libraries(mpm_tests PRIVATE mpm_core)
  add_test(NAME core_math_and_simulation COMMAND mpm_tests)
endif()
